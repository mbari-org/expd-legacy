C++ code for calculating ROV CTD depth from ROV CTD pressure

/************************
 function: calcDepth
 Depth in meters

 Ref: Saunders & Fofnoff, DSR, 23, 109-111 (1976)
      Parameters were re-fit for the 1980 equation
      of state for seawater (EOS80)

 Variables :
    P  - pressure in dbars  (not SI)
    L  - latitude in degrees

************************/
void CTD::calcDepth (float P, float L)
{
   const static float deg2rad = M_PI/180;

   float phi = L * deg2rad;
   float sinPhi  = sin( phi );
   float sinPhi2= sinPhi*sinPhi;

   static const float c0 =  5.2788e-3;
   static const float c1 =  2.36e-5;
   static const float c3 =  1.092e-6;

   float gr = 9.780318 * (1.0 + (c0 + c1*sinPhi2)*sinPhi2) + c3 * P;

   const static float c4 = -1.82e-15;
   const static float c5 =  2.279e-10;
   const static float c6 =  2.2512e-5;
   const static float c7 =  9.72659;

   depth = (((c4*P + c5) * P - c6) * P + c7) * P;
   depth = depth / gr;

   if (Nan(depth) != 0) depth = missingValue;
   return;
}
